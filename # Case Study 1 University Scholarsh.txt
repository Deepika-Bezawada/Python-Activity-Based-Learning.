# Case Study 1: University Scholarship Allocation
applicants = [
    ("Asha", [88, 91, 84], 92, 250000),
    ("Bharat", [78, 90, 86], 90, 220000),
    ("Charan", [95, 88, 91], 82, 180000),
    ("Divya", [82, 85, 80], 88, 280000),
]

eligible = []
rejected = []

for name, marks, attendance, income in applicants:
    percentage = sum(marks) / len(marks)
    reasons = []
    if percentage < 80:
        reasons.append("aggregate below 80%")
    if min(marks) < 70:
        reasons.append("a subject mark below 70")
    if attendance < 85:
        reasons.append("attendance below 85%")
    if income > 300000:
        reasons.append("income above limit")

    if reasons:
        rejected.append((name, ", ".join(reasons)))
    else:
        eligible.append((name, percentage, attendance, income))

eligible.sort(key=lambda item: (item[1], item[2]), reverse=True)

print("SCHOLARSHIP RANK LIST")
for rank, item in enumerate(eligible, start=1):
    print(f"{rank}. {item[0]} - {item[1]:.2f}% aggregate, {item[2]}% attendance")
print("\nNOT ELIGIBLE")
for name, reason in rejected:
    print(f"{name}: {reason}")


#output
SCHOLARSHIP RANK LIST
1. Asha - 87.67% aggregate, 92% attendance
2. Bharat - 84.67% aggregate, 90% attendance
3. Divya - 82.33% aggregate, 88% attendance

NOT ELIGIBLE
Charan: attendance below 85%
