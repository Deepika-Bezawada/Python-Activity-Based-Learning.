# Case Study 7: Warehouse Inventory Reconciliation
# Tuple: (product, opening, incoming, dispatched, recorded_closing)
records = [
    ("Pens", 50, 30, 45, 35),
    ("Notebooks", 80, 20, 70, 25),
    ("Folders", 40, 10, 25, 28),
]
target_stock = 20

print("INVENTORY RECONCILIATION")
for product, opening, incoming, dispatched, closing in records:
    expected = opening + incoming - dispatched
    difference = closing - expected
    reorder = max(target_stock - closing, 0)

    print(f"\nProduct: {product}")
    print(f"Expected closing stock: {expected}")
    print(f"Recorded closing stock: {closing}")
    if difference == 0:
        print("Status: Matched")
    else:
        print(f"Status: DISCREPANCY ({difference:+d} units)")
    print(f"Replenishment required: {reorder} units")
