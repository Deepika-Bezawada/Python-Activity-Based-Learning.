# Case Study 9: Password Policy Validation
passwords = ["Hello@123", "abc123", "AAAhello1!", "MyPassword9!"]

for password in passwords:
    failures = []
    if len(password) < 8:
        failures.append("must contain at least 8 characters")
    if not any(ch.isupper() for ch in password):
        failures.append("needs an uppercase letter")
    if not any(ch.islower() for ch in password):
        failures.append("needs a lowercase letter")
    if not any(ch.isdigit() for ch in password):
        failures.append("needs a digit")
    if not any(not ch.isalnum() for ch in password):
        failures.append("needs a special character")

    repeated = False
    for i in range(len(password) - 2):
        if password[i] == password[i + 1] == password[i + 2]:
            repeated = True
    if repeated:
        failures.append("must not contain three identical consecutive characters")
    if "password" in password.lower():
        failures.append("contains prohibited word 'password'")

    print(f"\nPassword sample: {password}")
    if failures:
        print("INVALID")
        for failure in failures:
            print("-", failure)
    else:
        print("VALID: all policy checks passed")


#output

VALID: all policy checks passed

- needs an uppercase letter
- needs a special character

INVALID
